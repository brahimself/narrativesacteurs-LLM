\chapter{Annexes}

\section{Checklist conformite Livrable 2 (UE Chef d'oeuvre)}
\begin{itemize}[leftmargin=2em]
  \item Le document annonce explicitement ``Livrable 2''.
  \item Le contenu couvre toutes les experimentations realisees.
  \item Le protocole d'evaluation est precise et reproductible.
  \item Les resultats sont compares (baseline + modeles fine-tunes).
  \item Les limites et perspectives sont explicites.
  \item Le lien avec l'application de demonstration est documente.
\end{itemize}

\section{Checklist soutenance (20 min + 15 min)}
\subsection*{Plan recommande (20 min)}
\begin{itemize}[leftmargin=2em]
  \item 0--2 min: probleme, objectifs, contribution.
  \item 2--6 min: donnees et pipeline (collecte, chunking, extraction, traduction FR->EN, paires \texttt{en/fr\_en}).
  \item 6--11 min: apprentissage similarite (paires, fine-tuning, protocoles).
  \item 11--15 min: resultats (baseline vs allMiniLM FT vs MPNET FT).
  \item 15--18 min: demo Spaces (generation, seuil, retries).
  \item 18--20 min: limites, perspectives, conclusion.
\end{itemize}

\subsection*{Questions probables du jury}
\begin{itemize}[leftmargin=2em]
  \item Pourquoi certains scores sont proches de 1.00?
  \item Comment avez-vous choisi et interprete le seuil?
  \item Quel est le gain reel du fine-tuning versus baseline?
  \item Quelle robustesse sur des cas difficiles (hard negatives)?
  \item Quels biais des donnees Wikipedia et impacts potentiels?
\end{itemize}
